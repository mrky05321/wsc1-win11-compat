cmake_minimum_required(VERSION 3.20)
project(wsc1_win11_compat VERSION 1.0.3 LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Use MSVC and -A Win32; the game and backend are x86")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(wsc1_ddraw SHARED src/proxy.cpp src/ddraw.def)
set_target_properties(wsc1_ddraw PROPERTIES OUTPUT_NAME ddraw)
target_compile_features(wsc1_ddraw PRIVATE cxx_std_11)
target_compile_options(wsc1_ddraw PRIVATE /W4 /utf-8)
target_compile_definitions(wsc1_ddraw PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0602)
target_link_libraries(wsc1_ddraw PRIVATE user32 winmm comctl32)

option(WSC1_BUILD_WINDOW_TESTS "Build the native window behavior regression checks" OFF)
if(WSC1_BUILD_WINDOW_TESTS)
  enable_testing()
  add_executable(wsc1_window_tests tests/window_behavior.cpp)
  target_compile_features(wsc1_window_tests PRIVATE cxx_std_11)
  target_compile_options(wsc1_window_tests PRIVATE /W4 /utf-8)
  target_compile_definitions(wsc1_window_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0602)
  target_link_libraries(wsc1_window_tests PRIVATE user32 comctl32)
  add_test(NAME window_behavior COMMAND wsc1_window_tests)
  add_executable(wsc1_cd_manifest_tests tests/cd_manifest.cpp)
  target_compile_features(wsc1_cd_manifest_tests PRIVATE cxx_std_11)
  target_compile_options(wsc1_cd_manifest_tests PRIVATE /W4 /utf-8)
  target_compile_definitions(wsc1_cd_manifest_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0602)
  add_test(NAME cd_manifest COMMAND wsc1_cd_manifest_tests)
endif()
